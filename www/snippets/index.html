Singine · SilkPage · sindoc

Composable bash commands from singine, silkpage, and sindoc repositories

Commands

Install singine

singine Makefile / README
# Install singine CLI into ~/.local/bin
make install          # sh-compatible
make install-bash     # bash-specific
make install-workstation  # + git-filter-repo

# After install, source the dotfile:
source ~/.singinerc
singine --help

# Or run without installing:
/usr/bin/python3 -m singine serve
/usr/bin/python3 -m singine cv pdf --output ~/Downloads/cv.pdf

singine serve — start the SilkPage appserver

singine bin/SingineServe.java
# Start the Java HTTP appserver (XSLT on demand)
singine serve                       # port 4242, silkpage www/
singine serve --port 8080           # custom port
singine serve --webroot /path/to/www

# Or directly:
java bin/SingineServe.java 4242 /path/to/www

# Check status:
curl http://localhost:4242/api/cv/status

# Download CV formats via API:
curl 'http://localhost:4242/api/cv/download?format=pdf' -o cv.pdf
curl 'http://localhost:4242/api/cv/download?format=rtf' -o cv.rtf

singine cv — CV generation

singine singine/cv.py
singine cv html                     # print path to cv/index.html
singine cv md                       # print path to cv-sina-heshmati.md
singine cv rtf --output ~/cv.rtf    # generate RTF (pure Python)
singine cv pdf --output ~/cv.pdf    # PDF via Chrome headless
singine cv serve --port 4242        # start XSLT server

singine install baremetal

singine singine/command.py
# Full machine bootstrap: pip install, dotfile, launcher, Node upgrade
singine install baremetal

# What it does:
# 1. python3 -m pip install -e /path/to/singine  (editable)
# 2. Creates ~/.singinerc  (PATH, aliases, env vars)
# 3. Adds  source ~/.singinerc  to ~/.bash_profile
# 4. Installs ~/.local/bin/singine  launcher
# 5. brew upgrade node  (background)

# Apply to current shell:
source ~/.singinerc
which singine

singine graph publish — Logseq graph to HTML

singine singine/graph.py
# Publish ~/ws/ls/kern to www/graph/
singine graph publish
singine graph publish --graph ~/ws/ls/kern
singine graph publish --out /path/to/www/graph

# Output:
#   www/graph/index.html         landing page
#   www/graph/pages/*.html       individual pages
#   www/graph/journals/*.html    journal entries
#   www/graph/graph-data.xml     RDF/XML graph manifest

singine mime xml — MIME-to-XML tree

singine singine/mime_xml.py
singine mime xml --root ~/ws/singine/main/singine --out /tmp/tree.xml
singine mime html                   # show MIME capabilities table
singine mime file path/to/file.py   # wrap single file as XML

SilkPage build — Ant / XProc pipeline

silkpage build.sh / Makefile
# Build the SilkPage site (requires Ant + Saxon)
make docs-html         # HTML docs output
make website-preview   # Ant-based preview
./build.sh             # XProc/Calabash full pipeline

# Dependencies check:
make check
# → checks: make xsltproc xmllint tidy zip java javac ant

# XSLT pipeline:
#   layout.xml → autolayout.xml  (Saxon 12 HE via Calabash)
#   index.xml + main.xsl → dist/website/  (Saxon 6.5.3 Docbook-Website)

Build Mule Runtime CE (Collibra Connect legacy)

mule (fork) build.sh
# In ~/ws/mule/fork/mule/
./build.sh                        # fast: skip tests, style, GPG
./build.sh -pl core               # single module
./build.sh -DskipTests=false      # with unit tests

# Requirements: Java 17+ (SDKMAN), Maven 3.9+
# build.sh auto-patches BOM POMs and clears .lastUpdated locks
